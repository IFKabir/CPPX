62template <typename T> Node<T> *BinaryTree<T>::clone_subtree(const Node<T> *node)
63{
64 if (!node)
65 return nullptr;
66
67 struct StackEntry
68 {
69 const Node<T> *src;
70 Node<T> *dest;
71 int stage;
72 };
73
74 Node<T> *new_root = m_pool.allocate(node->m_data);
75 new_root->m_height = node->m_height;
76 new_root->m_color = node->m_color;
77
78 std::stack<StackEntry> s;
79 s.push({node, new_root, 0});
80
81 while (!s.empty())
82 {
83 auto &top = s.top();
84 if (top.stage == 0)
85 {
86 top.stage = 1;
87 if (top.src->p_left)
88 {
89 Node<T> *child = m_pool.allocate(top.src->p_left->m_data);
90 child->m_height = top.src->p_left->m_height;
91 child->m_color = top.src->p_left->m_color;
92 child->p_parent = top.dest;
93 top.dest->p_left = child;
94 s.push({top.src->p_left, child, 0});
95 }
96 }
97 else if (top.stage == 1)
98 {
99 top.stage = 2;
100 if (top.src->p_right)
101 {
102 Node<T> *child = m_pool.allocate(top.src->p_right->m_data);
103 child->m_height = top.src->p_right->m_height;
104 child->m_color = top.src->p_right->m_color;
105 child->p_parent = top.dest;
106 top.dest->p_right = child;
107 s.push({top.src->p_right, child, 0});
108 }
109 }
110 else
111 {
112 s.pop();
113 }
114 }
115
116 return new_root;
117}
